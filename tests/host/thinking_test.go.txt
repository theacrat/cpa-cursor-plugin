package claude

import (
	"encoding/json"
	"os"
	"testing"

	"github.com/router-for-me/CLIProxyAPI/v7/sdk/pluginapi"
)

func TestCursorNativeThinkingTranslation(t *testing.T) {
	original := []byte(`{"model":"model","messages":[{"role":"user","content":"hi"}],"thinking":{"type":"enabled"}}`)
	payload := ConvertClaudeRequestToOpenAI("model", original, false)
	req := pluginapi.ExecutorRequest{Model: "model", SourceFormat: "chat-completions", Payload: payload, OriginalRequest: original}
	raw, err := json.Marshal(req)
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(os.Getenv("CPA_THINKING_FIXTURE"), raw, 0o600); err != nil {
		t.Fatal(err)
	}
}
