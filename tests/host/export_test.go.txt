func TestExportHostFixture(t *testing.T) {
	now := time.Now().UTC().Truncate(time.Second)
	s := NewAuthService(DefaultConfig())
	defer s.Close()
	s.now = func() time.Time { return now }
	raw, _ := json.Marshal(map[string]string{"type": "cursor", "access_token": fixtureJWT("host-fixture", now.Add(10*time.Minute)), "api_key": "fixture-key"})
	parsed, err := s.ParseAuth(context.Background(), pluginapi.AuthParseRequest{RawJSON: raw})
	if err != nil {
		t.Fatal(err)
	}
	payload, err := json.Marshal(parsed.Auth)
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(os.Getenv("CPA_HOST_FIXTURE"), payload, 0o600); err != nil {
		t.Fatal(err)
	}
	var req pluginapi.ExecutorRequest
	if err := json.Unmarshal(mustReadHostFixture(t, os.Getenv("CPA_THINKING_FIXTURE")), &req); err != nil {
		t.Fatal(err)
	}
	if got := requestEffort(req); got != "thinking" {
		t.Fatalf("translated native thinking resolved to %q", got)
	}
	e := NewExecutor(DefaultConfig())
	defer e.Close()
	metadata := modelWireNumber(modelWireString(nil, 1, "review-gpt"), 15, 200000)
	metadata = modelWireNumber(metadata, 10, 1)
	metadata = modelWireNumber(metadata, 5, 1)
	fast := modelWireNumber(modelWireString(nil, 1, "review-gpt-fast"), 15, 100000)
	models := discoverMetadata(t, e, "metadata-account", "", hostLabelRows(), availableFixture(metadata, fast))
	encoded, err := json.Marshal(models)
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(os.Getenv("CPA_MODEL_FIXTURE"), encoded, 0o600); err != nil {
		t.Fatal(err)
	}
}

func TestRestoreHostModelListMetadata(t *testing.T) {
	e := NewExecutor(DefaultConfig())
	defer e.Close()
	metadata := modelWireNumber(modelWireString(nil, 1, "review-gpt"), 15, 200000)
	metadata = modelWireNumber(metadata, 10, 1)
	metadata = modelWireNumber(metadata, 5, 1)
	fast := modelWireNumber(modelWireString(nil, 1, "review-gpt-fast"), 15, 100000)
	discoverMetadata(t, e, "metadata-account", "", hostLabelRows(), availableFixture(metadata, fast))
	response, err := e.InterceptResponse(context.Background(), pluginapi.ResponseInterceptRequest{Body: mustReadHostFixture(t, os.Getenv("CPA_MODEL_LIST_FIXTURE")), StatusCode: 200})
	if err != nil {
		t.Fatal(err)
	}
	var root struct {
		Data []map[string]any `json:"data"`
	}
	if err := json.Unmarshal(response.Body, &root); err != nil {
		t.Fatal(err)
	}
	found, foundFast := false, false
	for _, row := range root.Data {
		if row["id"] == "cursor/review-gpt-fast" {
			foundFast = true
			if row["display_name"] != "Review GPT (Fast)" || row["context_length"] != float64(100000) {
				t.Fatalf("host list lost fast label or metadata: %v", row)
			}
			continue
		}
		if row["id"] != "cursor/review-gpt" {
			continue
		}
		found = true
		if row["context_length"] != float64(200000) || row["tool_call"] != true || row["reasoning"] != true || row["modalities"] == nil || row["display_name"] != "Review GPT (cursor)" {
			t.Fatalf("host list metadata missing: %v", row)
		}
	}
	if !found || !foundFast {
		t.Fatal("host list lost namespaced Cursor speed tiers")
	}
}

func hostLabelRows() []*pb.ModelDetails {
	return []*pb.ModelDetails{
		{ModelId: "review-gpt-low", DisplayName: "Review GPT Low"},
		{ModelId: "review-gpt-high", DisplayName: "Review GPT High"},
		{ModelId: "review-gpt-low-fast", DisplayName: "Review GPT Low Fast"},
		{ModelId: "review-gpt-high-fast", DisplayName: "Review GPT High Fast"},
	}
}

func mustReadHostFixture(t *testing.T, path string) []byte {
	t.Helper()
	raw, err := os.ReadFile(path)
	if err != nil {
		t.Fatal(err)
	}
	return raw
}
